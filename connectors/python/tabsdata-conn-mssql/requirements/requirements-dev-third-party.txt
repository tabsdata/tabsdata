# Docker-driven end-to-end tests for the MSSQL connector. `pyodbc` itself
# is pip-installable, but it requires `unixodbc` and the Microsoft ODBC
# Driver for SQL Server (`msodbcsql18`) to be installed at the system
# level (e.g. `brew install unixodbc msodbcsql18` on macOS). A missing
# driver surfaces as a clear `pyodbc.InterfaceError` at engine build time.
docker==7.1.0
pyodbc==5.3.0
